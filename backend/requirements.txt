# --- security (Milestone 2 consumes these; installed now so the foundation is complete) ---
# PyJWT, not python-jose. python-jose carries an algorithm-confusion advisory
# (PYSEC-2024-232) plus two JWT-bomb DoS advisories, and the 3.4.0 that fixes
# them pins pyasn1<0.5.0 - which cannot coexist with google-auth below, whose
# pyasn1-modules needs >=0.6.1. PyJWT has none of those advisories and no
# pyasn1 dependency. The whole migration was four lines in app/core/security.py.
PyJWT==2.10.1
argon2-cffi==23.1.0
passlib==1.7.4

# --- push notifications ---
# FCM's HTTP v1 API authenticates with a Google service account, not the
# retired server key, so a signed JWT exchange is required. google-auth is the
# smallest thing that does it - firebase-admin pulls in the whole Firebase SDK
# for one token mint.
google-auth==2.35.0

# google-auth ships the credential types but NOT a transport. Its
# `google.auth.transport.requests` module imports this at call time, and
# `requests` is an optional extra nobody installs by default - so the failure
# appears only on a clean deploy, never on a laptop that already had it.
requests==2.33.0

# --- http client ---
# Used by the push sender and by the Brevo transport. Was a test-only
# dependency; it is a runtime one now.
